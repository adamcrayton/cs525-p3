## Design

Being that every address is a uint32_t in host byte order, so all of the subnet math is bit operations on the mask. cidr_mask() builds the mask whereas everything else derives from that. The network is addr & mask, the broadcast is addr | ~mask, and a block contains x when (x & mask) is equal to (addr & mask).

/0 (0xFFFFFFFF << (32 - prefix)) is a shift by 32 when the prefix is 0, and shifting a 32-bit value by its full width is undefined behavior in C. On x86 it usually gives back 0xFFFFFFF insead of 0, making the default route match only one address. cidr_mask() return 0 before it ever shifts.

/31 is a point-to-point link (RFC 3021), so both addresses are hosts and neither is a network or bradcoast address. cidr_first_host() returns the network address instead of network + 1, cidr_last_host() returns the broadcast address instead of braodcast - 1, and cidr_host_count() returns 2.

/32 is a single host, so it's network, broadcast, first host and last host all the same address and the host count is 1. /31 and /32 share the prefix >= 31 branch in the first and last host functions.

#include <string.h>

/** Prefixes over 32 are treated as 32 everywhere, the way cidr_mask() says. */
static uint8_t clamp_prefix(uint8_t prefix) { return prefix > 32 ? 32 : prefix; }

/**
 * Parse a decimal number of 1 to max_digits digits starting at *p, with no
 * sign and no leading zero (a lone "0" is fine). On success *p is moved past
 * the digits. Nothing after the digits is checked; the caller does that.
 */
static bool parse_decimal(const char **p, unsigned max_digits, unsigned max_value,
                          unsigned *value) {
  const char *s = *p;
  unsigned digits = 0;
  unsigned v = 0;

  while (s[digits] >= '0' && s[digits] <= '9') {
    if (digits == max_digits) {
      return false; /* too many digits */
    }
    v = v * 10 + (unsigned)(s[digits] - '0');
    digits++;
  }
  if (digits == 0) {
    return false; /* empty field, e.g. "1..2.3" or "+1" */
  }
  if (digits > 1 && s[0] == '0') {
    return false; /* leading zero, e.g. "010" */
  }
  if (v > max_value) {
    return false;
  }
  *value = v;
  *p = s + digits;
  return true;
}

/**
 * Parse a dotted quad at the start of text. On success *end points at the
 * first character after the fourth octet, which the caller must check.
 */
static bool parse_quad(const char *text, const char **end, uint32_t *addr) {
  const char *p = text;
  uint32_t result = 0;

  for (int i = 0; i < 4; i++) {
    unsigned octet;
    if (i > 0) {
      if (*p != '.') {
        return false;
      }
      p++;
    }
    if (!parse_decimal(&p, 3, 255, &octet)) {
      return false;
    }
    result = (result << 8) | octet;
  }
  *end = p;
  *addr = result;
  return true;
}

/* ------------------------------------------------------------------------
 * Task 1 - Address parsing and formatting
 * --------------------------------------------------------------------- */

bool ip_parse(const char *text, uint32_t *addr) {
  const char *end;
  uint32_t result;

  if (text == NULL || addr == NULL) {
    return false;
  }
  if (!parse_quad(text, &end, &result) || *end != '\0') {
    return false; /* malformed, or junk after the last octet */
  }
  *addr = result;
  return true;
}

void ip_format(uint32_t addr, char out[IP_STR_LEN]) {
  if (out == NULL) {
    return;
  }
  snprintf(out, IP_STR_LEN, "%u.%u.%u.%u", (unsigned)(addr >> 24) & 0xFFu,
           (unsigned)(addr >> 16) & 0xFFu, (unsigned)(addr >> 8) & 0xFFu,
           (unsigned)addr & 0xFFu);
}

bool cidr_parse(const char *text, cidr_t *block) {
  const char *p;
  uint32_t addr;
  unsigned prefix;

  if (text == NULL || block == NULL) {
    return false;
  }
  if (!parse_quad(text, &p, &addr) || *p != '/') {
    return false; /* bad address, or the slash is missing */
  }
  p++;
  if (!parse_decimal(&p, 2, 32, &prefix) || *p != '\0') {
    return false; /* missing, too large, leading zero, or trailing junk */
  }
  block->addr = addr;
  block->prefix = (uint8_t)prefix;
  return true;
}

void cidr_format(cidr_t block, char out[CIDR_STR_LEN]) {
  uint8_t prefix = clamp_prefix(block.prefix);
  size_t len;

  if (out == NULL) {
    return;
  }
  /* The dotted quad is at most 15 characters, and "/32" at most 3 more, so
   * this always fits in CIDR_STR_LEN. The prefix is written by hand because
   * the compiler can not see that it has at most two digits. */
  ip_format(block.addr, out);
  len = strlen(out);
  out[len++] = '/';
  if (prefix >= 10) {
    out[len++] = (char)('0' + prefix / 10);
  }
  out[len++] = (char)('0' + prefix % 10);
  out[len] = '\0';
}

/* ------------------------------------------------------------------------
 * Task 2 - Subnet arithmetic
 * --------------------------------------------------------------------- */

uint32_t cidr_mask(uint8_t prefix) {
  prefix = clamp_prefix(prefix);
  if (prefix == 0) {
    return 0; /* 0xFFFFFFFF << 32 would be undefined behavior */
  }
  return UINT32_MAX << (32 - prefix);
}

uint32_t cidr_network(cidr_t block) { return block.addr & cidr_mask(block.prefix); }

uint32_t cidr_broadcast(cidr_t block) { return block.addr | ~cidr_mask(block.prefix); }

uint32_t cidr_first_host(cidr_t block) {
  uint32_t network = cidr_network(block);
  return clamp_prefix(block.prefix) >= 31 ? network : network + 1;
}

uint32_t cidr_last_host(cidr_t block) {
  uint32_t broadcast = cidr_broadcast(block);
  return clamp_prefix(block.prefix) >= 31 ? broadcast : broadcast - 1;
}

uint64_t cidr_host_count(cidr_t block) {
  uint8_t prefix = clamp_prefix(block.prefix);
  if (prefix == 32) {
    return 1;
  }
  if (prefix == 31) {
    return 2;
  }
  return (UINT64_C(1) << (32 - prefix)) - 2;
}

bool cidr_contains(cidr_t block, uint32_t addr) {
  uint32_t mask = cidr_mask(block.prefix);
  return (addr & mask) == (block.addr & mask);
}

bool cidr_is_network(cidr_t block) { return (block.addr & ~cidr_mask(block.prefix)) == 0; }

bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out) {
  uint8_t prefix = clamp_prefix(block.prefix);
  unsigned bits = 0;
  unsigned new_prefix;

  if (out == NULL || !cidr_is_network(block)) {
    return false;
  }
  if (n == 0 || (n & (n - 1)) != 0) {
    return false; /* not a power of two */
  }
  while ((UINT32_C(1) << bits) != n) {
    bits++; /* bits = log2(n), at most 31 */
  }
  new_prefix = prefix + bits;
  if (new_prefix > 32 || i >= n) {
    return false;
  }
  /*
   * Subnet i starts i subnet-sizes above the network address. The shift is
   * done in 64 bits because i << 32 (n = 1 on a /0) is undefined for 32.
   */
  out->addr = block.addr + (uint32_t)((uint64_t)i << (32 - new_prefix));
  out->prefix = (uint8_t)new_prefix;
  return true;
}

 * Task 3 - Longest prefix match
 * --------------------------------------------------------------------- */

/**
 * The table is a growable array of routes. Lookup is a linear scan that keeps
 * the longest matching prefix, so insertion order never matters. Destinations
 * are unique (table_add() enforces it), so there is never a tie to break.
 */
struct fwd_table {
  route_t *routes;
  size_t count;
  size_t capacity;
};

/** Space or tab: the field separators in a route line. */
static bool is_blank(char c) { return c == ' ' || c == '\t'; }

/**
 * Copy the next blank-separated field of *p into buf (size bufsize) and move
 * *p past it. Fails if there is no field or it does not fit.
 */
static bool next_field(const char **p, const char *end, char *buf, size_t bufsize) {
  const char *s = *p;
  size_t len = 0;

  while (s < end && is_blank(*s)) {
    s++;
  }
  while (s + len < end && !is_blank(s[len])) {
    len++;
  }
  if (len == 0 || len >= bufsize) {
    return false;
  }
  memcpy(buf, s, len);
  buf[len] = '\0';
  *p = s + len;
  return true;
}

bool route_parse(const char *line, route_t *route) {
  /* Field buffers: one longer than the longest valid field, so overlong
   * fields are caught by next_field() rather than silently truncated. */
  char dest[CIDR_STR_LEN];
  char gw[IP_STR_LEN];
  char dev[DEV_NAME_LEN];
  const char *start;
  const char *end;
  route_t r;

  if (line == NULL || route == NULL) {
    return false;
  }

  /* Trim leading and trailing whitespace, including the newline. */
  start = line;
  while (*start != '\0' && isspace((unsigned char)*start)) {
    start++;
  }
  end = start + strlen(start);
  while (end > start && isspace((unsigned char)end[-1])) {
    end--;
  }

  if (!next_field(&start, end, dest, sizeof dest) ||
      !next_field(&start, end, gw, sizeof gw) ||
      !next_field(&start, end, dev, sizeof dev)) {
    return false;
  }
  if (start != end) {
    return false; /* a fourth field */
  }

  memset(&r, 0, sizeof r);

  if (strcmp(dest, "default") == 0) {
    r.dest.addr = 0;
    r.dest.prefix = 0;
  } else if (!cidr_parse(dest, &r.dest) || !cidr_is_network(r.dest)) {
    return false;
  }

  if (strcmp(gw, "direct") == 0) {
    r.gateway = 0;
  } else if (!ip_parse(gw, &r.gateway)) {
    return false;
  }

  /* The trim and the separators cover spaces and tabs, but a stray '\r' or
   * other control character inside a field would end up in the name. */
  for (const char *c = dev; *c != '\0'; c++) {
    if (!isgraph((unsigned char)*c)) {
      return false;
    }
  }
  memcpy(r.dev, dev, sizeof r.dev);

  *route = r;
  return true;
}

fwd_table_t *table_create(void) {
  return calloc(1, sizeof(fwd_table_t)); /* empty: NULL array, 0 count */
}

void table_destroy(fwd_table_t *table) {
  if (table == NULL) {
    return;
  }
  free(table->routes);
  free(table);
}

size_t table_count(const fwd_table_t *table) { return table == NULL ? 0 : table->count; }

bool table_add(fwd_table_t *table, const route_t *route) {
  route_t copy;

  if (table == NULL || route == NULL || !cidr_is_network(route->dest)) {
    return false;
  }
  for (size_t k = 0; k < table->count; k++) {
    const cidr_t *d = &table->routes[k].dest;
    if (d->addr == route->dest.addr &&
        clamp_prefix(d->prefix) == clamp_prefix(route->dest.prefix)) {
      return false; /* duplicate destination */
    }
  }
  if (table->count == table->capacity) {
    size_t new_capacity = table->capacity == 0 ? 8 : table->capacity * 2;
    route_t *grown = NULL;
    /* Guard the multiplication against overflow; then grown stays NULL. */
    if (new_capacity <= SIZE_MAX / sizeof(route_t)) {
      grown = lab_realloc(table->routes, new_capacity * sizeof(route_t));
    }
    if (grown == NULL) {
      return false; /* out of memory: the old array is unchanged */
    }
    table->routes = grown;
    table->capacity = new_capacity;
  }

  copy = *route;
  copy.dest.prefix = clamp_prefix(copy.dest.prefix);
  copy.dev[DEV_NAME_LEN - 1] = '\0'; /* never trust the caller's terminator */
  table->routes[table->count++] = copy;
  return true;
}

bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line) {
  char *line = NULL;
  size_t cap = 0;
  size_t line_no = 0;
  bool ok = true;

  if (bad_line != NULL) {
    *bad_line = 0;
  }
  if (table == NULL || in == NULL) {
    return false;
  }

  while (getline(&line, &cap, in) != -1) {
    char *hash;
    const char *p;
    route_t route;

    line_no++;
    hash = strchr(line, '#');
    if (hash != NULL) {
      *hash = '\0'; /* drop the comment */
    }
    for (p = line; *p != '\0' && isspace((unsigned char)*p); p++) {
    }
    if (*p == '\0') {
      continue; /* blank once the comment is gone */
    }
    if (!route_parse(line, &route) || !table_add(table, &route)) {
      if (bad_line != NULL) {
        *bad_line = line_no;
      }
      ok = false;
      break;
    }
  }

  if (ok && ferror(in)) {
    ok = false; /* read error; *bad_line stays 0 */
  }
  free(line);
  return ok;
}

const route_t *table_lookup(const fwd_table_t *table, uint32_t dest) {
  const route_t *best = NULL;

  if (table == NULL) {
    return NULL;
  }
  for (size_t k = 0; k < table->count; k++) {
    const route_t *r = &table->routes[k];
    if (cidr_contains(r->dest, dest) && (best == NULL || r->dest.prefix > best->dest.prefix)) {
      best = r;
    }
  }
  return best;
}

#ifndef LAB_H
#define LAB_H

#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>

/*
 * Every address in this file is a uint32_t in host byte order, so the first
 * octet of 192.168.1.10 is the most significant byte: 0xC0A8010A.
 */

/** Room for the longest dotted quad, "255.255.255.255", and its '\0'. */
#define IP_STR_LEN 16

/** Room for the longest CIDR string, "255.255.255.255/32", and its '\0'. */
#define CIDR_STR_LEN 19

/** Room for an interface name; IFNAMSIZ on Linux, including the '\0'. */
#define DEV_NAME_LEN 16

/** An address block in CIDR notation: an address and a prefix length. */
typedef struct {
  uint32_t addr;  /**< any address in the block, host byte order */
  uint8_t prefix; /**< 0 to 32 */
} cidr_t;

/** One entry in a forwarding table. */
typedef struct {
  cidr_t dest;            /**< destination block; host bits must be zero */
  uint32_t gateway;       /**< next hop, or 0 when the block is directly connected */
  char dev[DEV_NAME_LEN]; /**< outgoing interface, for example "eth0" */
} route_t;

/** A forwarding table. Its layout is private to lab.c. */
typedef struct fwd_table fwd_table_t;

/* ------------------------------------------------------------------------
 * Task 1 - Address parsing and formatting
 * --------------------------------------------------------------------- */

/**
 * @brief Parse a dotted quad such as "192.168.1.10".
 *
 * Exactly four decimal octets from 0 to 255 separated by single periods, with
 * nothing before, between or after them. An octet with a leading zero ("010")
 * is rejected, because inet_aton() reads it as octal and that is never what a
 * person typing it meant.
 *
 * @param text The string to parse.
 * @param addr Where the address is stored on success; untouched on failure.
 * @return true if text is a valid dotted quad, false otherwise (including NULL).
 */
bool ip_parse(const char *text, uint32_t *addr);

/**
 * @brief Format an address as a dotted quad.
 * @param addr The address, host byte order.
 * @param out A buffer of at least IP_STR_LEN bytes.
 */
void ip_format(uint32_t addr, char out[IP_STR_LEN]);

/**
 * @brief Parse a block in CIDR notation such as "192.168.1.0/24".
 *
 * The address follows the rules of ip_parse() and the prefix is a decimal
 * number from 0 to 32 with no leading zero. The slash and the prefix are
 * required. Host bits may be set ("132.178.227.11/25" is an interface
 * address), so use cidr_is_network() when they must not be.
 *
 * @param text The string to parse.
 * @param block Where the block is stored on success; untouched on failure.
 * @return true if text is valid CIDR notation, false otherwise (including NULL).
 */
bool cidr_parse(const char *text, cidr_t *block);

/**
 * @brief Format a block as "a.b.c.d/p", using the address exactly as stored.
 *
 * A prefix over 32 is written as 32, the way cidr_mask() treats it.
 *
 * @param block The block.
 * @param out A buffer of at least CIDR_STR_LEN bytes.
 */
void cidr_format(cidr_t block, char out[CIDR_STR_LEN]);

/* ------------------------------------------------------------------------
 * Task 2 - Subnet arithmetic
 * --------------------------------------------------------------------- */

/**
 * @brief The subnet mask for a prefix length: /24 is 255.255.255.0.
 *
 * Careful with /0. Shifting a 32 bit value by 32 is undefined behavior in C.
 *
 * @param prefix 0 to 32; anything larger is treated as 32.
 * @return The mask, host byte order.
 */
uint32_t cidr_mask(uint8_t prefix);

/** @brief The lowest address in the block: the address with the host bits cleared. */
uint32_t cidr_network(cidr_t block);

/** @brief The highest address in the block: the address with the host bits set. */
uint32_t cidr_broadcast(cidr_t block);

/**
 * @brief The first address that can be given to a host.
 *
 * For /0 to /30 that is the network address plus one. A /31 is a point to
 * point link (RFC 3021) where both addresses are usable, and a /32 is a single
 * host, so for those two the first host is the network address.
 */
uint32_t cidr_first_host(cidr_t block);

/**
 * @brief The last address that can be given to a host.
 *
 * The broadcast address minus one for /0 to /30, and the broadcast address
 * itself for /31 and /32 (see cidr_first_host()).
 */
uint32_t cidr_last_host(cidr_t block);

/**
 * @brief How many addresses in the block can be given to hosts.
 *
 * 2^(32 - prefix) - 2 for /0 to /30, 2 for /31, and 1 for /32. For a /0 the
 * 2^32 in that formula needs 33 bits, more than a uint32_t holds, which is why
 * this returns a uint64_t.
 */
uint64_t cidr_host_count(cidr_t block);

/** @brief true if addr falls inside the block. */
bool cidr_contains(cidr_t block, uint32_t addr);

/** @brief true if the block's host bits are all zero, so it names a network. */
bool cidr_is_network(cidr_t block);

/**
 * @brief The i-th of n equally sized subnets of a block.
 *
 * Splitting a /24 into 4 gives four /26 blocks, and subnet 0 is the lowest.
 * Equal sizes are only possible when n is a power of two, and the new prefix
 * can not pass 32.
 *
 * @param block The block to split; it must be a network (cidr_is_network()).
 * @param n How many subnets, a power of two from 1 up.
 * @param i Which subnet, 0 to n - 1.
 * @param out Where the subnet is stored on success; untouched on failure.
 * @return false if out is NULL, the block is not a network, n is not a power
 *         of two, the split would need a prefix longer than 32, or i is out of
 *         range.
 */
bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out);

/* ------------------------------------------------------------------------
 * Task 3 - Longest prefix match
 * --------------------------------------------------------------------- */

/**
 * @brief Parse one line of a forwarding table file.
 *
 * A line is three fields separated by spaces or tabs:
 *
 *     <destination> <gateway> <device>
 *
 * The destination is a block in CIDR notation with no host bits set, or the
 * word "default" for 0.0.0.0/0. The gateway is a dotted quad, or the word
 * "direct" for a directly connected block (stored as 0). The device is 1 to
 * DEV_NAME_LEN - 1 characters. Leading and trailing whitespace, including a
 * trailing newline, is ignored. Comments are not handled here.
 *
 * @param line The line.
 * @param route Where the route is stored on success; untouched on failure.
 * @return true if the line is a valid route, false otherwise (including NULL).
 */
bool route_parse(const char *line, route_t *route);

/**
 * @brief Create an empty forwarding table.
 * @return The table, which the caller frees with table_destroy(), or NULL if
 *         memory could not be allocated.
 */
fwd_table_t *table_create(void);

/** @brief Free a table and everything in it. NULL is allowed and does nothing. */
void table_destroy(fwd_table_t *table);

/** @brief The number of routes in the table; 0 for NULL. */
size_t table_count(const fwd_table_t *table);

/**
 * @brief Add a copy of a route to the table, growing it as needed.
 * @return false if either argument is NULL, the destination has host bits
 *         set, the table already holds a route to the same block, or memory
 *         could not be allocated. The table is unchanged on failure.
 */
bool table_add(fwd_table_t *table, const route_t *route);

/**
 * @brief Read a forwarding table file and add every route in it.
 *
 * Everything from a '#' to the end of a line is a comment, and a line that is
 * blank once the comment is removed is skipped. Every other line must be a
 * valid route (route_parse()) that table_add() accepts.
 *
 * @param table The table to add to.
 * @param in The open file to read.
 * @param bad_line If not NULL, set to the 1-based number of the first line
 *        that failed, or 0 if the arguments were NULL or the file could not
 *        be read.
 * @return true if every line loaded. On false, routes from the lines before
 *         the bad one stay in the table.
 */
bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line);

/**
 * @brief Longest prefix match: the route whose block contains dest and has the
 *        longest prefix.
 *
 * A default route (0.0.0.0/0) matches everything, so it is chosen only when
 * nothing more specific does. The order the routes were added in does not
 * matter.
 *
 * @return The matching route, which stays owned by the table, or NULL if no
 *         route matches (or table is NULL).
 */
const route_t *table_lookup(const fwd_table_t *table, uint32_t dest);

#endif // LAB_H

#include <errno.h>
#include <stdlib.h>

#ifndef TEST

#define PROG "myapp"

  EXIT_OK = 0,
  EXIT_USAGE = 1,
  EXIT_TABLE = 2,
  EXIT_UNREACHABLE = 3,

static const char USAGE[] = "Usage: myapp info <address>/<prefix>\n"
                            "       myapp split <network>/<prefix> <count>\n"

  fprintf(stderr, PROG ": %s\n%s", what, USAGE);
  return EXIT_USAGE;

/**
 * Parse a subnet count: decimal, no sign, no leading zero, fits in uint32_t.
 * Whether it is a power of two is checked by the caller.
 */
static bool parse_count(const char *text, uint32_t *count) {
  uint64_t v = 0;

  if (text[0] == '\0' || (text[0] == '0' && text[1] != '\0')) {
    return false;
  }
  for (const char *p = text; *p != '\0'; p++) {
    if (*p < '0' || *p > '9') {
      return false;
    }
    v = v * 10 + (uint64_t)(*p - '0');
    if (v > UINT32_MAX) {
      return false;
    }
  }
  *count = (uint32_t)v;
  return true;
}

static int cmd_info(int argc, char *argv[]) {
  cidr_t block;
  char buf[IP_STR_LEN];

  if (argc != 3) {

  printf("network    %s\n", buf);

  printf("broadcast  %s\n", buf);
  ip_format(cidr_first_host(block), buf);
  printf("first host %s\n", buf);
  ip_format(cidr_last_host(block), buf);

  printf("hosts      %llu\n", (unsigned long long)cidr_host_count(block));
  return EXIT_OK;

 * split

static int cmd_split(int argc, char *argv[]) {
  cidr_t block;
  uint32_t count;
  unsigned bits = 0;

  if (argc != 4) {
    return usage_error("split takes a <network>/<prefix> and a <count>");
  }
  if (!cidr_parse(argv[2], &block)) {
    fprintf(stderr, PROG ": %s: not a valid <network>/<prefix>\n", argv[2]);
    return EXIT_USAGE;
  }
  if (!cidr_is_network(block)) {
    char net[CIDR_STR_LEN];
    cidr_t fixed = {cidr_network(block), block.prefix};
    cidr_format(fixed, net);
    fprintf(stderr, PROG ": %s has host bits set, so it is not a network (did you mean %s?)\n",
            argv[2], net);
    return EXIT_USAGE;
  }
  if (!parse_count(argv[3], &count) || count == 0 || (count & (count - 1)) != 0) {
    fprintf(stderr, PROG ": %s: the count must be a power of two (1, 2, 4, 8, ...)\n", argv[3]);
    return EXIT_USAGE;
  }
  while ((UINT32_C(1) << bits) != count) {
    bits++;
  }
  if (block.prefix + bits > 32) {
    fprintf(stderr, PROG ": a /%u can not be split into %s subnets: they would need a /%u\n",
            (unsigned)block.prefix, argv[3], (unsigned)block.prefix + bits);
    return EXIT_USAGE;
  }

  for (uint32_t i = 0; i < count; i++) {
    cidr_t sub;
    char buf[CIDR_STR_LEN];
    if (!cidr_subnet(block, count, i, &sub)) {
      fprintf(stderr, PROG ": could not compute subnet %u\n", (unsigned)i); /* not reached */
      return EXIT_USAGE;
    }
    cidr_format(sub, buf);
    puts(buf);
  }
  return EXIT_OK;
}

 * lookup
 * --------------------------------------------------------------------- */

static int cmd_lookup(int argc, char *argv[]) {
  const char *path;
  FILE *in;
  fwd_table_t *table;
  size_t bad_line = 0;
  bool malformed = false;
  bool unreachable = false;

  if (argc < 4) {
    return usage_error("lookup takes a <table-file> and at least one <address>");
  }
  path = argv[2];

  in = fopen(path, "r");
  if (in == NULL) {
    fprintf(stderr, PROG ": %s: %s\n", path, strerror(errno));
    return EXIT_TABLE;
  }
  table = table_create();
  if (table == NULL) {
    fclose(in);
    fprintf(stderr, PROG ": out of memory\n");
    return EXIT_TABLE;
  }
  if (!table_load(table, in, &bad_line)) {
    if (bad_line > 0) {
      fprintf(stderr, PROG ": %s:%zu: not a valid route\n", path, bad_line);
    } else {
      fprintf(stderr, PROG ": %s: could not read the file\n", path);
    }
    fclose(in);
    table_destroy(table);
    return EXIT_TABLE;
  }
  fclose(in);

  for (int k = 3; k < argc; k++) {
    uint32_t addr;
    const route_t *route;
    char buf[IP_STR_LEN];

    if (!ip_parse(argv[k], &addr)) {
      fprintf(stderr, PROG ": %s: not a valid address\n", argv[k]);
      malformed = true;
      continue;
    }
    ip_format(addr, buf);
    route = table_lookup(table, addr);
    if (route == NULL) {
      printf("%s unreachable\n", buf);
      unreachable = true;
    } else if (route->gateway != 0) {
      char gw[IP_STR_LEN];
      ip_format(route->gateway, gw);
      printf("%s via %s dev %s\n", buf, gw, route->dev);
    } else {
      printf("%s dev %s\n", buf, route->dev);
    }
  }

  table_destroy(table);
  if (malformed) {
    return EXIT_USAGE;
  }
  return unreachable ? EXIT_UNREACHABLE : EXIT_OK;
}

  }
  if (strcmp(argv[1], "lookup") == 0) {
    return cmd_lookup(argc, argv);
  }
  fprintf(stderr, PROG ": unknown command '%s'\n%s", argv[1], USAGE);
  return EXIT_USAGE;
}

void test_cidr_subnet_edges(void) {
  cidr_t s;
  char buf[CIDR_STR_LEN];

  /* n = 1 is the block itself, even for /0 (no shift by 32) */
  TEST_ASSERT_TRUE(cidr_subnet(blk("0.0.0.0/0"), 1, 0, &s));
  cidr_format(s, buf);
  TEST_ASSERT_EQUAL_STRING("0.0.0.0/0", buf);

  /* /0 into halves */
  TEST_ASSERT_TRUE(cidr_subnet(blk("0.0.0.0/0"), 2, 1, &s));
  cidr_format(s, buf);
  TEST_ASSERT_EQUAL_STRING("128.0.0.0/1", buf);

  /* /0 into 2^31 /31s: the last one */
  TEST_ASSERT_TRUE(cidr_subnet(blk("0.0.0.0/0"), UINT32_C(0x80000000), UINT32_C(0x7FFFFFFF), &s));
  cidr_format(s, buf);
  TEST_ASSERT_EQUAL_STRING("255.255.255.254/31", buf);

  /* /30 into four /32s */
  TEST_ASSERT_TRUE(cidr_subnet(blk("10.0.0.4/30"), 4, 3, &s));
  cidr_format(s, buf);
  TEST_ASSERT_EQUAL_STRING("10.0.0.7/32", buf);

  /* /32 into one */
  TEST_ASSERT_TRUE(cidr_subnet(blk("10.0.0.7/32"), 1, 0, &s));
}

void test_route_parse_rejects(void) {
  const char *bad[] = {
      "",
      "   \n",
      "10.0.0.0/8 direct",                       /* two fields */
      "10.0.0.0/8",                              /* one field */
      "10.0.0.0/8 direct eth0 extra",            /* four fields */
      "10.0.0.1/8 direct eth0",                  /* host bits set */
      "10.0.0.0 direct eth0",                    /* no prefix */
      "10.0.0.0/33 direct eth0",                 /* bad prefix */
      "10.0.0.0/8 10.0.0 eth0",                  /* bad gateway */
      "10.0.0.0/8 010.0.0.1 eth0",               /* octal-looking gateway */
      "10.0.0.0/8 direct abcdefghijklmnop",      /* 16 chars: too long */
      "Default direct eth0",                     /* keywords are exact */
      "default Direct eth0",
      "10.0.0.0/8 direct eth0 # comment",        /* comments are not handled here */
      "10.0.0.0/8,direct,eth0",
  };
  for (size_t k = 0; k < sizeof bad / sizeof bad[0]; k++) {
    route_t r;
    memset(&r, 0x5A, sizeof r);
    TEST_ASSERT_FALSE_MESSAGE(route_parse(bad[k], &r), bad[k]);
    TEST_ASSERT_EQUAL_HEX32_MESSAGE(0x5A5A5A5A, r.gateway, "route must be untouched");
  }
  TEST_ASSERT_FALSE(route_parse(NULL, &(route_t){0}));
}

void test_table_basics(void) {
  route_t r;
  TEST_ASSERT_NOT_NULL(table);
  TEST_ASSERT_EQUAL_size_t(0, table_count(table));
  TEST_ASSERT_EQUAL_size_t(0, table_count(NULL));
  TEST_ASSERT_NULL(table_lookup(table, IP(1, 2, 3, 4)));
  TEST_ASSERT_NULL(table_lookup(NULL, IP(1, 2, 3, 4)));
  table_destroy(NULL);

  TEST_ASSERT_TRUE(route_parse("10.0.0.0/8 direct eth0", &r));
  TEST_ASSERT_FALSE(table_add(NULL, &r));
  TEST_ASSERT_FALSE(table_add(table, NULL));
  TEST_ASSERT_TRUE(table_add(table, &r));
  TEST_ASSERT_FALSE(table_add(table, &r)); /* duplicate */
  TEST_ASSERT_EQUAL_size_t(1, table_count(table));

  /* same block, different gateway: still a duplicate */
  TEST_ASSERT_TRUE(route_parse("10.0.0.0/8 1.1.1.1 eth9", &r));
  TEST_ASSERT_FALSE(table_add(table, &r));

  /* same address, different prefix: a different block */
  TEST_ASSERT_TRUE(route_parse("10.0.0.0/16 direct eth0", &r));
  TEST_ASSERT_TRUE(table_add(table, &r));

  /* host bits set */
  r.dest = (cidr_t){IP(10, 0, 0, 1), 8};
  TEST_ASSERT_FALSE(table_add(table, &r));
  TEST_ASSERT_EQUAL_size_t(2, table_count(table));
}

/* Defined in lab.c when it is built with -DTEST; see lab_realloc() there. */
extern bool lab_test_fail_next_alloc;

void test_table_add_out_of_memory(void) {
  char line[64];
  route_t r;

  /* The first add allocates room for 8; fill it so the 9th must grow. */
  for (int k = 0; k < 8; k++) {
    snprintf(line, sizeof line, "10.0.%d.0/24 direct eth%d", k, k);
    add(line);
  }
  TEST_ASSERT_TRUE(route_parse("10.0.8.0/24 direct eth8", &r));

  lab_test_fail_next_alloc = true;
  TEST_ASSERT_FALSE(table_add(table, &r));
  TEST_ASSERT_FALSE(lab_test_fail_next_alloc); /* the failure was used */

  /* Unchanged on failure: same count, old routes still there. */
  TEST_ASSERT_EQUAL_size_t(8, table_count(table));
  TEST_ASSERT_NULL(table_lookup(table, IP(10, 0, 8, 1)));
  TEST_ASSERT_EQUAL_STRING("eth7", table_lookup(table, IP(10, 0, 7, 1))->dev);

  /* And it recovers once memory is available again. */
  TEST_ASSERT_TRUE(table_add(table, &r));
  TEST_ASSERT_EQUAL_size_t(9, table_count(table));
  TEST_ASSERT_EQUAL_STRING("eth8", table_lookup(table, IP(10, 0, 8, 1))->dev);
}
